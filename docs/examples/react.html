⚛️ Parallax Visualizer - React Examples

🚀 Basic Usage

The simplest way to use Parallax Visualizer in a React application. Just import the component and pass your desired props. 
import React from 'react';
import { ParallaxVisualizer } from 'parallax-visualizer/react';

function App() {
    return (
        <div className="app">
            <h1>My Music Visualizer</h1>
            <ParallaxVisualizer
                mode="bars"
                audioSource="microphone"
                lolcat={true}
                style={{ width: '100%', height: '400px' }}
            />
        </div>
    );
}

export default App;

🎛️ Advanced Usage with Controls

A more advanced example with mode switching, audio source controls, and event handling. 
import React, { useState, useRef } from 'react';
import { ParallaxVisualizer, VisualizerControls } from 'parallax-visualizer/react';

function MusicPlayer() {
    const [mode, setMode] = useState('bars');
    const [isPlaying, setIsPlaying] = useState(false);
    const [audioSource, setAudioSource] = useState('microphone');
    const visualizerRef = useRef();

    const handleStart = () => {
        visualizerRef.current?.start();
        setIsPlaying(true);
    };

    const handleStop = () => {
        visualizerRef.current?.stop();
        setIsPlaying(false);
    };

    const handleModeChange = (newMode) => {
        setMode(newMode);
        visualizerRef.current?.setMode(newMode);
    };

    const handleFileUpload = (file) => {
        visualizerRef.current?.switchSource({
            type: 'file',
            file: file
        });
        setAudioSource('file');
    };

    return (
        <div className="music-player">
            {/* Controls Component */}
            <VisualizerControls
                currentMode={mode}
                isPlaying={isPlaying}
                onModeChange={handleModeChange}
                onStart={handleStart}
                onStop={handleStop}
                onFileUpload={handleFileUpload}
            />

            {/* Visualizer */}
            <ParallaxVisualizer
                ref={visualizerRef}
                mode={mode}
                audioSource={audioSource}
                lolcat={true}
                adaptiveQuality={true}
                style={{ height: '500px' }}
                onSourceChange={(source) => console.log('Source:', source)}
                onModeChange={(mode) => console.log('Mode:', mode)}
            />
        </div>
    );
}

export default MusicPlayer;

🪝 Using Hooks

Use the useParallaxVisualizer hook for more granular control over the visualization. 
import React, { useEffect } from 'react';
import { useParallaxVisualizer } from 'parallax-visualizer/react';

function CustomVisualizer() {
    const {
        containerRef,
        visualizer,
        isPlaying,
        currentMode,
        audioIntensity,
        start,
        stop,
        setMode,
        setLolcat
    } = useParallaxVisualizer({
        mode: 'matrix',
        audioSource: 'microphone',
        lolcat: true
    });

    // React to audio intensity changes
    useEffect(() => {
        if (audioIntensity > 0.8) {
            // Switch to more intense visualization
            setMode('hypercube');
        } else if (audioIntensity < 0.3) {
            // Switch to calmer visualization
            setMode('wave');
        }
    }, [audioIntensity, setMode]);

    return (
        <div className="custom-visualizer">
            {/* Audio intensity indicator */}
            <div className="intensity-bar">
                <div 
                    className="intensity-fill"
                    style={{ width: `${audioIntensity * 100}%` }}
                />
            </div>

            {/* Current mode display */}
            <p>Current Mode: {currentMode}</p>

            {/* Controls */}
            <button onClick={isPlaying ? stop : start}>
                {isPlaying ? 'Stop' : 'Start'}
            </button>

            {/* Visualizer container */}
            <div 
                ref={containerRef}
                style={{ height: '400px', border: '1px solid #333' }}
            />
        </div>
    );
}

export default CustomVisualizer;

🌐 Context Provider

Share visualizer state across multiple components using the VisualizerProvider. 
import React from 'react';
import { 
    VisualizerProvider, 
    useVisualizerContext 
} from 'parallax-visualizer/react';

// Child component that uses the context
function VisualizerStats() {
    const { audioIntensity, fps, currentMode } = useVisualizerContext();

    return (
        <div className="stats">
            <p>Audio Intensity: {(audioIntensity * 100).toFixed(1)}%</p>
            <p>FPS: {fps}</p>
            <p>Mode: {currentMode}</p>
        </div>
    );
}

// Main app component
function App() {
    return (
        <VisualizerProvider
            mode="bars"
            audioSource="microphone"
            lolcat={true}
        >
            <div className="app">
                <VisualizerStats />
                {/* Other components can access visualizer state */}
            </div>
        </VisualizerProvider>
    );
}

export default App;

📋 Available Props & Methods

ParallaxVisualizer Component Props:
  • mode: Visualization mode (string)
  • audioSource: Audio input source ('microphone', 'file', 'stream')
  • lolcat: Enable rainbow colors (boolean)
  • adaptiveQuality: Enable performance optimization (boolean)
  • style: CSS styles object
  • className: CSS class name
  • onSourceChange: Audio source change callback
  • onModeChange: Mode change callback
  • onStart: Visualization start callback
  • onStop: Visualization stop callback
useParallaxVisualizer Hook Returns:
  • containerRef: React ref for the container element
  • visualizer: ParallaxVisualizer instance
  • isPlaying: Current playback state
  • currentMode: Current visualization mode
  • audioIntensity: Current audio intensity (0-1)
  • start(): Start visualization
  • stop(): Stop visualization
  • setMode(mode): Change visualization mode
  • setLolcat(enabled): Toggle rainbow colors
