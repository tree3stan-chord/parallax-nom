// Import the ParallaxVisualizer
import ParallaxVisualizer from 'parallax-visualizer';

// Create visualizer instance
const visualizer = new ParallaxVisualizer({
    container: '#visualizer',
    mode: 'bars',
    audioSource: 'microphone',
    lolcat: true,
    adaptiveQuality: true
});

// Start visualization
visualizer.start();

// Change visualization mode
visualizer.setMode('matrix');

// Toggle rainbow colors
visualizer.setLolcat(false);

// Switch audio source
visualizer.switchSource({
    type: 'file',
    file: audioFile
});

// Listen to events
visualizer.on('sourceChange', (newSource) => {
    console.log('Audio source changed to:', newSource);
});

// Clean up when done
visualizer.destroy();
